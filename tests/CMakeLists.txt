cmake_minimum_required(VERSION 3.15)
project(AppMetricaBridgeTests LANGUAGES CXX)

enable_testing()
find_package(Threads REQUIRED)
add_executable(appmetrica_inbox_test inbox_test.cpp)
target_compile_features(appmetrica_inbox_test PRIVATE cxx_std_11)
target_link_libraries(appmetrica_inbox_test PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(appmetrica_inbox_test PRIVATE /utf-8 /UNDEBUG)
else()
    target_compile_options(appmetrica_inbox_test PRIVATE -UNDEBUG)
endif()
add_test(NAME appmetrica_inbox_test COMMAND appmetrica_inbox_test)
